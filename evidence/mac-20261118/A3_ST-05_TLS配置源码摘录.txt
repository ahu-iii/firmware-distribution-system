   139	void TlsChannelSecurity::prepare(QSslSocket* socket, bool serverSide) {
   140	  Q_UNUSED(serverSide);
   141	  QSslConfiguration config = socket->sslConfiguration();
   142	
   143	  // 只允许 TLS 1.3，只启用一个带认证加密的套件（检查项 4-3、1-4）
   144	  config.setProtocol(QSsl::TlsV1_3);
   145	  config.setCiphers({cipher_});
   146	  // 密钥交换只用 X25519 临时密钥（前向安全）
   147	  config.setEllipticCurves({QSslEllipticCurve::fromLongName(QStringLiteral("X25519"))});
   148	
   149	  // 双向认证：VerifyPeer 表示对端必须出示证书并通过校验，客户端和服务端都这样设。
   150	  // 信任库只放自建 CA，系统证书库里的任何 CA 都不被信任（检查项 4-1）。
   151	  config.setPeerVerifyMode(QSslSocket::VerifyPeer);
   152	  config.setPeerVerifyDepth(1);  // 链上只有“叶子 + 根”，不接受更长的链
   153	  config.setCaCertificates({caCert_});
   154	  config.setLocalCertificate(localCert_);
   155	  config.setPrivateKey(localKey_);
   156	
   157	  // 不做会话复用和 0-RTT：每次连接完整握手、重新协商密钥（协议第 2 节）
   158	  config.setSslOption(QSsl::SslOptionDisableSessionTickets, true);
   159	  config.setSslOption(QSsl::SslOptionDisableSessionSharing, true);
   160	  config.setSslOption(QSsl::SslOptionDisableSessionPersistence, true);
   161	  config.setSslOption(QSsl::SslOptionDisableCompression, true);
   162	  socket->setSslConfiguration(config);
